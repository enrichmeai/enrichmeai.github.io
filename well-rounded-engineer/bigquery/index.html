Guide / Cloud providers / BigQuery

BigQuery SQL

GoogleSQL patterns, cost controls and the bq CLI. On-demand pricing bills bytes scanned, so most tips are about scanning less.

Partitioned, clustered table

CREATE TABLE IF NOT EXISTS `proj.analytics.orders` (
  order_id    STRING NOT NULL,
  order_ref   STRING,
  store_id    STRING,
  order_date  DATE,
  lines       ARRAY<STRUCT<sku STRING, qty INT64>>,
  _landed_at  TIMESTAMP
)
PARTITION BY DATE_TRUNC(order_date, MONTH)
CLUSTER BY store_id, order_ref
OPTIONS (
  require_partition_filter = TRUE,
  description = 'Orders, one row per order_id',
  labels = [('layer', 'analytics')]
);

-- integer range partition
PARTITION BY RANGE_BUCKET(order_year, GENERATE_ARRAY(1995, 2031, 1))
-- ingestion-time partition
PARTITION BY _PARTITIONDATE

Limit: 10,000 partitions per table. Daily partitions over 30 years (about 11,000 days) overflow it; use MONTH or YEAR.

Deduplicate with QUALIFY

-- latest version of each key
SELECT *
FROM `proj.raw.orders`
WHERE order_year = 2024
QUALIFY ROW_NUMBER() OVER (
  PARTITION BY store_id, order_ref
  ORDER BY _landed_at DESC
) = 1;

-- latest event per entity (append-only ledger → current state)
SELECT * EXCEPT(rn) FROM (
  SELECT *, ROW_NUMBER() OVER (PARTITION BY run_id, stage
                               ORDER BY event_ts DESC) rn
  FROM `proj.ops.job_run_events`)
WHERE rn = 1;

-- whole row of the max, no window
SELECT ARRAY_AGG(t ORDER BY order_date DESC LIMIT 1)[OFFSET(0)].*
FROM `proj.analytics.orders` t GROUP BY store_id;

MERGE (upsert)

MERGE `proj.mart.serving` T
USING (
  SELECT * FROM `proj.analytics.orders`
  WHERE order_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
) S
ON T.order_id = S.order_id
   AND T.order_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)  -- prune T
WHEN MATCHED AND T.row_hash != S.row_hash THEN
  UPDATE SET order_ref = S.order_ref, order_date = S.order_date, row_hash = S.row_hash
WHEN NOT MATCHED BY TARGET THEN
  INSERT ROW
WHEN NOT MATCHED BY SOURCE AND T.order_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY) THEN
  DELETE;

-- change detection hash
TO_HEX(SHA256(TO_JSON_STRING(STRUCT(order_ref, store_id, order_date)))) AS row_hash
FARM_FINGERPRINT(CONCAT(store_id, '|', order_ref))   -- INT64 key

Dates & safe parsing

PARSE_DATE('%d-%m-%Y', '05-05-1959')          -- errors on bad input
SAFE.PARSE_DATE('%d-%m-%Y', raw)               -- NULL on bad input
SAFE_CAST(order_seq AS INT64)

-- never let a parse failure hide as a NULL: count it
SELECT COUNTIF(raw IS NOT NULL AND SAFE.PARSE_DATE('%d-%m-%Y', raw) IS NULL) AS unparsed
FROM t;

DATE_TRUNC(d, MONTH)   DATE_ADD(d, INTERVAL 1 DAY)   DATE_DIFF(a, b, DAY)
LAST_DAY(d, MONTH)     EXTRACT(YEAR FROM d)          FORMAT_DATE('%Y%m%d', d)
TIMESTAMP_TRUNC(ts, HOUR)  DATETIME(ts, 'Asia/Kolkata')
CURRENT_DATE('Asia/Kolkata')
UNIX_MILLIS(ts)  TIMESTAMP_MICROS(n)

-- calendar spine for gap finding
SELECT d FROM UNNEST(GENERATE_DATE_ARRAY('2025-01-01', CURRENT_DATE())) d
LEFT JOIN facts f ON f.dt = d WHERE f.dt IS NULL;

ARRAY, STRUCT, UNNEST, JSON

-- explode
SELECT o.order_ref, i.sku, i.qty
FROM `proj.analytics.orders` o, UNNEST(o.lines) i;      -- comma = CROSS JOIN
-- keep rows with empty arrays
LEFT JOIN UNNEST(o.lines) i

-- build
SELECT order_ref, ARRAY_AGG(STRUCT(sku, qty) ORDER BY sku) lines
FROM order_lines GROUP BY order_ref;

ARRAY_LENGTH(a)  a[SAFE_OFFSET(0)]  ARRAY_TO_STRING(a, ',')
'x' IN UNNEST(a)  EXISTS(SELECT 1 FROM UNNEST(a) v WHERE v > 3)

-- JSON
JSON_VALUE(payload, '$.store.code')            -- scalar as STRING
JSON_QUERY(payload, '$.lines')                  -- JSON fragment
JSON_QUERY_ARRAY(payload, '$.lines')            -- ARRAY<JSON> on a JSON column, ARRAY<STRING> on a STRING
LAX_INT64(payload.count)                       -- on a JSON column
TO_JSON_STRING(STRUCT(a, b))
PARSE_JSON(s)

Window functions

SELECT
  store, order_date, n,
  SUM(n) OVER (PARTITION BY store ORDER BY order_date)          AS running,
  AVG(n) OVER (PARTITION BY store ORDER BY order_date
               ROWS BETWEEN 6 PRECEDING AND CURRENT ROW)           AS avg_7d,
  LAG(n) OVER (PARTITION BY store ORDER BY order_date)          AS prev,
  n / SUM(n) OVER (PARTITION BY store)                             AS share,
  RANK() OVER (PARTITION BY store ORDER BY n DESC)                 AS rnk,
  NTILE(4) OVER (ORDER BY n)                                       AS quartile
FROM daily;

-- named window
SELECT SUM(n) OVER w, MAX(n) OVER w FROM daily
WINDOW w AS (PARTITION BY store ORDER BY order_date);

APPROX_COUNT_DISTINCT(order_ref)   APPROX_QUANTILES(n, 100)[OFFSET(50)]

Load, export, external

LOAD DATA INTO `proj.raw.orders`
FROM FILES (format = 'PARQUET',
            uris = ['gs://landing/data/year=2024/*.parquet']);

EXPORT DATA OPTIONS (
  uri = 'gs://exports/serving/2025-01-31/*.parquet',   -- one * required
  format = 'PARQUET', overwrite = TRUE, compression = 'SNAPPY'
) AS SELECT * FROM `proj.mart.serving` WHERE _run_date = '2025-01-31';

CREATE OR REPLACE EXTERNAL TABLE `proj.raw.orders_ext`
WITH PARTITION COLUMNS
OPTIONS (format = 'PARQUET',
         uris = ['gs://landing/data/*'],
         hive_partition_uri_prefix = 'gs://landing/data');

Pipe syntax

FROM `proj.analytics.orders`
|> WHERE order_date >= '2024-01-01'
|> AGGREGATE COUNT(*) AS n GROUP BY store_id
|> ORDER BY n DESC
|> LIMIT 10;

-- other operators: SELECT, EXTEND (add columns), SET, DROP, RENAME,
-- JOIN, AS (alias), PIVOT, UNPIVOT, TABLESAMPLE, CALL

Reads top to bottom in execution order. Mixes freely with standard SQL in the same project.

Metadata & cost queries

-- most expensive queries in the last day
SELECT user_email, job_id,
       ROUND(total_bytes_billed / POW(1024, 4), 3) AS tib_billed,
       total_slot_ms, LEFT(query, 120) AS q
FROM `region-asia-south1`.INFORMATION_SCHEMA.JOBS
WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 1 DAY)
  AND job_type = 'QUERY'
ORDER BY total_bytes_billed DESC LIMIT 20;

-- partitions and their row counts
SELECT partition_id, total_rows, total_logical_bytes
FROM `proj.analytics.INFORMATION_SCHEMA.PARTITIONS`
WHERE table_name = 'orders' ORDER BY partition_id;

-- columns of a table
SELECT column_name, data_type, is_nullable
FROM `proj.analytics.INFORMATION_SCHEMA.COLUMNS` WHERE table_name = 'orders';

-- table sizes
SELECT table_id, row_count, size_bytes / 1e9 AS gb
FROM `proj.analytics.__TABLES__` ORDER BY size_bytes DESC;

Time travel, snapshots, wildcards

-- read the table as it was (up to 7 days back by default)
SELECT * FROM `proj.analytics.orders`
FOR SYSTEM_TIME AS OF TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 1 HOUR);

-- undo a bad write
CREATE OR REPLACE TABLE `proj.analytics.orders_restore` AS
SELECT * FROM `proj.analytics.orders` FOR SYSTEM_TIME AS OF '2025-01-31 06:00:00+00';

-- zero-copy snapshot / clone
CREATE SNAPSHOT TABLE `proj.backup.orders_20250131` CLONE `proj.analytics.orders`;
CREATE TABLE `proj.dev.orders` CLONE `proj.analytics.orders`;

-- sharded tables
SELECT * FROM `proj.logs.events_*` WHERE _TABLE_SUFFIX BETWEEN '20250101' AND '20250131';

Cost & performance rules

  • Name columns. Storage is columnar; SELECT * bills every column. SELECT * EXCEPT(raw_html) drops wide ones.
  • Filter on the partition column with a constant or parameter. A filter through a function of another table's subquery may not prune.
  • LIMIT does not reduce bytes on a regular table. Use table preview or TABLESAMPLE SYSTEM (1 PERCENT).
  • Dry-run first and set maximum_bytes_billed on jobs that could run away.
  • Cluster on high-cardinality filter/join columns; order matters, first column filters best.
  • Join big to small after filtering; pre-aggregate before joining.
  • Avoid row-by-row DML. Batch into one MERGE: each statement is a query you pay for, and only two UPDATE, DELETE or MERGE statements per table run at once (20 more can queue; beyond that they fail).
  • Materialised views for repeated aggregates; BI Engine for dashboards.
  • Execution graph in the console shows skew, spills and which stage costs the slots.

bq CLI

bq query --use_legacy_sql=false --dry_run 'SELECT ...'
bq query --use_legacy_sql=false --format=prettyjson --max_rows=20 \
  --parameter=from_d:DATE:2024-01-01 \
  'SELECT COUNT(*) FROM `proj.analytics.orders` WHERE order_date >= @from_d'
bq query --nouse_legacy_sql --maximum_bytes_billed=10000000000 < q.sql

bq ls proj:analytics
bq show --schema --format=prettyjson proj:analytics.orders
bq head -n 10 proj:analytics.orders

bq load --source_format=PARQUET --replace \
  'proj:raw.events$20250131' 'gs://landing/events/dt=2025-01-31/*.parquet'   # one partition
bq extract --destination_format=PARQUET proj:mart.serving 'gs://exp/serving-*.parquet'

bq cp proj:analytics.orders proj:backup.orders_20250131
bq mk --table --time_partitioning_field=order_date --clustering_fields=store_id \
  proj:analytics.t schema.json
bq ls -j -a -n 20          # recent jobs
bq show -j <job_id>

Levels: what to expect

1 Foundation

  • Query with named columns and partition filters
  • Dry-run and read bytes processed
  • Load data from GCS

2 Practitioner

  • Design partitioned and clustered tables
  • Write MERGE, scripting and parameterised queries
  • Use INFORMATION_SCHEMA to find cost and usage

3 Senior

  • Optimise slot time and cost across a project
  • Choose pricing models and reservations
  • Design security: datasets, authorised views, row and column policies

4 Expert

  • Own the warehouse platform and its cost governance
  • Integrate BigQuery with streaming, ML and the wider lakehouse
  • Advise on architecture across many teams

Each level includes the ones before it. See Start here for what the levels mean.
