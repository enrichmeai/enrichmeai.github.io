Partitioned, clustered table

CREATE TABLE IF NOT EXISTS `proj.fdp.judgment` (
  judgment_key STRING NOT NULL,
  cnr          STRING,
  court_code   STRING,
  decision_date DATE,
  parties      ARRAY<STRUCT<role STRING, name STRING>>,
  _landed_at   TIMESTAMP
)
PARTITION BY DATE_TRUNC(decision_date, MONTH)
CLUSTER BY court_code, cnr
OPTIONS (
  require_partition_filter = TRUE,
  description = 'FDP judgments, one row per judgment_key',
  labels = [('layer', 'fdp')]
);

-- integer range partition
PARTITION BY RANGE_BUCKET(data_year, GENERATE_ARRAY(1950, 2031, 1))
-- ingestion-time partition
PARTITION BY _PARTITIONDATE

Limit: 10,000 partitions per table. Daily partitions over 76 years overflow it; use MONTH or YEAR.

Deduplicate with QUALIFY

-- latest version of each key
SELECT *
FROM `proj.odp.hc_judgment`
WHERE data_year = 2024
QUALIFY ROW_NUMBER() OVER (
  PARTITION BY cnr, order_number
  ORDER BY _landed_at DESC
) = 1;

-- latest event per entity (append-only ledger → current state)
SELECT * EXCEPT(rn) FROM (
  SELECT *, ROW_NUMBER() OVER (PARTITION BY run_id, stage
                               ORDER BY event_ts DESC) rn
  FROM `proj.dp_control.pipeline_run_events`)
WHERE rn = 1;

-- whole row of the max, no window
SELECT ARRAY_AGG(t ORDER BY decision_date DESC LIMIT 1)[OFFSET(0)].*
FROM `proj.fdp.judgment` t GROUP BY court_code;

MERGE (upsert)

MERGE `proj.cdp.serving` T
USING (
  SELECT * FROM `proj.fdp.judgment`
  WHERE decision_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
) S
ON T.judgment_key = S.judgment_key
   AND T.decision_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)  -- prune T
WHEN MATCHED AND T.row_hash != S.row_hash THEN
  UPDATE SET cnr = S.cnr, decision_date = S.decision_date, row_hash = S.row_hash
WHEN NOT MATCHED BY TARGET THEN
  INSERT ROW
WHEN NOT MATCHED BY SOURCE AND T.decision_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY) THEN
  DELETE;

-- change detection hash
TO_HEX(SHA256(TO_JSON_STRING(STRUCT(cnr, court_code, decision_date)))) AS row_hash
FARM_FINGERPRINT(CONCAT(cnr, '|', order_number))   -- INT64 key

Dates & safe parsing

PARSE_DATE('%d-%m-%Y', '05-05-1959')          -- errors on bad input
SAFE.PARSE_DATE('%d-%m-%Y', raw)               -- NULL on bad input
SAFE_CAST(order_seq AS INT64)

-- never let a parse failure hide as a NULL: count it
SELECT COUNTIF(raw IS NOT NULL AND SAFE.PARSE_DATE('%d-%m-%Y', raw) IS NULL) AS unparsed
FROM t;

DATE_TRUNC(d, MONTH)   DATE_ADD(d, INTERVAL 1 DAY)   DATE_DIFF(a, b, DAY)
LAST_DAY(d, MONTH)     EXTRACT(YEAR FROM d)          FORMAT_DATE('%Y%m%d', d)
TIMESTAMP_TRUNC(ts, HOUR)  DATETIME(ts, 'Asia/Kolkata')
CURRENT_DATE('Asia/Kolkata')
UNIX_MILLIS(ts)  TIMESTAMP_MICROS(n)

-- calendar spine for gap finding
SELECT d FROM UNNEST(GENERATE_DATE_ARRAY('2025-01-01', CURRENT_DATE())) d
LEFT JOIN facts f ON f.dt = d WHERE f.dt IS NULL;

ARRAY, STRUCT, UNNEST, JSON

-- explode
SELECT j.cnr, p.role, p.name
FROM `proj.fdp.judgment` j, UNNEST(j.parties) p;      -- comma = CROSS JOIN
-- keep rows with empty arrays
LEFT JOIN UNNEST(j.parties) p

-- build
SELECT cnr, ARRAY_AGG(STRUCT(role, name) ORDER BY role) parties
FROM parties GROUP BY cnr;

ARRAY_LENGTH(a)  a[SAFE_OFFSET(0)]  ARRAY_TO_STRING(a, ',')
'x' IN UNNEST(a)  EXISTS(SELECT 1 FROM UNNEST(a) v WHERE v > 3)

-- JSON
JSON_VALUE(payload, '$.court.code')            -- scalar as STRING
JSON_QUERY(payload, '$.parties')               -- JSON fragment
JSON_QUERY_ARRAY(payload, '$.parties')         -- ARRAY<JSON> on a JSON column, ARRAY<STRING> on a STRING
LAX_INT64(payload.count)                       -- on a JSON column
TO_JSON_STRING(STRUCT(a, b))

Window functions

SELECT
  court, decision_date, n,
  SUM(n) OVER (PARTITION BY court ORDER BY decision_date)          AS running,
  AVG(n) OVER (PARTITION BY court ORDER BY decision_date
               ROWS BETWEEN 6 PRECEDING AND CURRENT ROW)           AS avg_7d,
  LAG(n) OVER (PARTITION BY court ORDER BY decision_date)          AS prev,
  n / SUM(n) OVER (PARTITION BY court)                             AS share,
  RANK() OVER (PARTITION BY court ORDER BY n DESC)                 AS rnk,
  NTILE(4) OVER (ORDER BY n)                                       AS quartile
FROM daily;

-- named window
SELECT SUM(n) OVER w, MAX(n) OVER w FROM daily
WINDOW w AS (PARTITION BY court ORDER BY decision_date);

APPROX_COUNT_DISTINCT(cnr)   APPROX_QUANTILES(n, 100)[OFFSET(50)]

Metadata & cost queries

-- most expensive queries in the last day
SELECT user_email, job_id,
       ROUND(total_bytes_billed / POW(1024, 4), 3) AS tib_billed,
       total_slot_ms, LEFT(query, 120) AS q
FROM `region-asia-south1`.INFORMATION_SCHEMA.JOBS
WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 1 DAY)
  AND job_type = 'QUERY'
ORDER BY total_bytes_billed DESC LIMIT 20;

-- partitions and their row counts
SELECT partition_id, total_rows, total_logical_bytes
FROM `proj.fdp.INFORMATION_SCHEMA.PARTITIONS`
WHERE table_name = 'judgment' ORDER BY partition_id;

-- columns of a table
SELECT column_name, data_type, is_nullable
FROM `proj.fdp.INFORMATION_SCHEMA.COLUMNS` WHERE table_name = 'judgment';

-- table sizes
SELECT table_id, row_count, size_bytes / 1e9 AS gb
FROM `proj.fdp.__TABLES__` ORDER BY size_bytes DESC;

Time travel, snapshots, wildcards

-- read the table as it was (up to 7 days back by default)
SELECT * FROM `proj.fdp.judgment`
FOR SYSTEM_TIME AS OF TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 1 HOUR);

-- undo a bad write
CREATE OR REPLACE TABLE `proj.fdp.judgment_restore` AS
SELECT * FROM `proj.fdp.judgment` FOR SYSTEM_TIME AS OF '2025-01-31 06:00:00+00';

-- zero-copy snapshot / clone
CREATE SNAPSHOT TABLE `proj.backup.judgment_20250131` CLONE `proj.fdp.judgment`;
CREATE TABLE `proj.dev.judgment` CLONE `proj.fdp.judgment`;

-- sharded tables
SELECT * FROM `proj.logs.events_*` WHERE _TABLE_SUFFIX BETWEEN '20250101' AND '20250131';

Cost & performance rules

  • Name columns. Storage is columnar; SELECT * bills every column. SELECT * EXCEPT(raw_html) drops wide ones.
  • Filter on the partition column with a constant or parameter. A filter through a function of another table's subquery may not prune.
  • LIMIT does not reduce bytes on a regular table. Use table preview or TABLESAMPLE SYSTEM (1 PERCENT).
  • Dry-run first and set maximum_bytes_billed on jobs that could run away.
  • Cluster on high-cardinality filter/join columns; order matters, first column filters best.
  • Join big to small after filtering; pre-aggregate before joining.
  • Avoid row-by-row DML. Batch into one MERGE; DML quotas and costs are per statement.
  • Materialised views for repeated aggregates; BI Engine for dashboards.
  • Execution graph in the console shows skew, spills and which stage costs the slots.

  --parameter=from_d:DATE:2024-01-01 \
