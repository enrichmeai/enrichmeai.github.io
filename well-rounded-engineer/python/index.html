match (structural pattern matching)

match event:
    case {"state": "SUCCEEDED", "rows": int(n)} if n == 0:
        warn("succeeded with zero rows")
    case {"state": "SUCCEEDED", "rows": int(n)}:
        ok(n)
    case {"state": "FAILED", "error": str(msg)}:
        fail(msg)
    case {"state": state}:
        unknown(state)
    case _:
        raise ValueError(f"bad event: {event!r}")

match path.split("/"):
    case ["data", y, *rest] if y.startswith("year="): ...

Patterns match structure and types; they do not do string prefix matching. Use a guard (if) for that.

Errors, logging, context managers

import logging, contextlib, time
log = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s")

try:
    load(uri)
except NotFound:
    log.warning("missing %s", uri)          # lazy formatting
except Exception:
    log.exception("load failed for %s", uri) # includes traceback
    raise
else:
    log.info("loaded %s", uri)

@contextlib.contextmanager
def timed(label):
    t = time.perf_counter()
    try: yield
    finally: log.info("%s took %.2fs", label, time.perf_counter() - t)

with timed("odp load"): run()

Concurrency

# Threads for blocking I/O (GCS, BQ, HTTP)
from concurrent.futures import ThreadPoolExecutor, as_completed
with ThreadPoolExecutor(max_workers=16) as ex:
    futs = {ex.submit(copy, uri): uri for uri in uris}
    for f in as_completed(futs):
        try: f.result()
        except Exception as e: log.error("%s: %s", futs[f], e)

# asyncio with structured cancellation (3.11)
import asyncio
async def main():
    async with asyncio.TaskGroup() as tg:
        tasks = [tg.create_task(fetch(u)) for u in urls]
    return [t.result() for t in tasks]
asyncio.run(main())

BigQuery Python client

from google.cloud import bigquery
client = bigquery.Client(project="my-proj", location="asia-south1")

job_cfg = bigquery.QueryJobConfig(query_parameters=[
    bigquery.ScalarQueryParameter("from_d", "DATE", "2024-01-01"),
    bigquery.ArrayQueryParameter("courts", "STRING", ["2_5", "14_25"]),
])
sql = """SELECT court, COUNT(*) n FROM `my-proj.fdp.judgment`
         WHERE decision_date >= @from_d AND court IN UNNEST(@courts)
         GROUP BY court"""
for row in client.query(sql, job_config=job_cfg).result():
    print(row.court, row["n"])

df = client.query(sql, job_config=job_cfg).to_dataframe()   # needs db-dtypes

# Dry run
dry = client.query(sql, job_config=bigquery.QueryJobConfig(
    query_parameters=job_cfg.query_parameters,   # same parameters as the real query
    dry_run=True, use_query_cache=False))
print(dry.total_bytes_processed / 1e9, "GB")

# Load parquet from GCS
load = client.load_table_from_uri(
    "gs://bkt/data/year=2024/*.parquet", "my-proj.odp.hc_judgment",
    job_config=bigquery.LoadJobConfig(
        source_format=bigquery.SourceFormat.PARQUET,
        write_disposition=bigquery.WriteDisposition.WRITE_APPEND))
load.result()                     # raises on failure
print(load.output_rows)

client.insert_rows_json("my-proj.ops.events", [{"k": "v"}])  # returns errors list

import apache_beam as beam
from apache_beam.options.pipeline_options import PipelineOptions

with beam.Pipeline(options=PipelineOptions()) as p:
    (p
     | beam.io.ReadFromBigQuery(query="SELECT court FROM `p.odp.hc`",
                                use_standard_sql=True)
     | beam.Map(lambda r: (r["court"], 1))
     | beam.CombinePerKey(sum)
     | beam.Map(lambda kv: {"court": kv[0], "n": kv[1]})
     | beam.io.WriteToBigQuery("p:cdp.court_counts",
           schema="court:STRING,n:INTEGER",
           write_disposition=beam.io.BigQueryDisposition.WRITE_TRUNCATE))

Levels: what to expect

1 Foundation

  • Write functions, comprehensions and dataclasses
  • Use virtual environments and install packages
  • Write simple pytest tests

2 Practitioner

  • Type hints, logging, error handling and context managers
  • Use the BigQuery and GCS clients in pipelines
  • Package a project with pyproject.toml

3 Senior

  • Choose between threads, processes and asyncio
  • Design reusable libraries with clear APIs and tests
  • Profile memory and CPU; optimise data handling

4 Expert

  • Understand CPython internals, the GIL and free-threading
  • Own the organisation's Python tooling and packaging
  • Mentor and set code standards

Each level includes the ones before it. See Start here for what the levels mean.
